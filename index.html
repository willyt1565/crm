Team CRM · Setup
1
Create free Supabase project
Go to supabase.com → New Project → name it npm-crm.
2
Disable email confirmations
Authentication → Settings → turn off Enable email confirmations → Save.
3
Run SQL
SQL Editor → New query → paste and run:
drop trigger if exists on_auth_user_created on auth.users; drop function if exists handle_new_user(); drop table if exists tasks cascade; drop table if exists leads cascade; drop table if exists profiles cascade; drop table if exists sms_templates cascade; create table profiles ( id uuid references auth.users primary key, full_name text default '',email text default '', created_at timestamptz default now() ); alter table profiles enable row level security; create policy "insert own" on profiles for insert with check (auth.uid()=id); create policy "view team" on profiles for select using (auth.role()='authenticated'); create policy "update own" on profiles for update using (auth.uid()=id); create table leads ( id text primary key,data jsonb, assigned_to uuid references profiles(id), created_by uuid,updated_at bigint ); alter table leads enable row level security; create policy "team leads" on leads for all using (auth.role()='authenticated') with check (auth.role()='authenticated'); create policy "anon insert leads" on leads for insert with check (true); create table tasks ( id text primary key,data jsonb, created_by uuid,updated_at bigint ); alter table tasks enable row level security; create policy "team tasks" on tasks for all using (auth.role()='authenticated') with check (auth.role()='authenticated'); create table sms_templates ( id text primary key,name text,body text, created_by uuid,created_at timestamptz default now() ); alter table sms_templates enable row level security; create policy "team templates" on sms_templates for all using (auth.role()='authenticated') with check (auth.role()='authenticated');
4
Enter credentials
Project Settings → API → copy URL and publishable key.
